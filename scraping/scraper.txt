const cheerio = require("cheerio");
const fs = require("fs");

const url = "https://www.allabolag.se/bransch-s%C3%B6k?q=Arkitekter";

async function scrapeCompanies() {
    try {
        const response = await fetch(url, {
            headers: {
                "User-Agent": 
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/139.0.0.0 Safari/537.36",
                "Accept": 
                "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8",
                "Accept-Language": 
                "sv-SE,sv;q=0.9,en-US;q=0.8,en;q=0.7"
            }
        });

        if (!response.ok) {
            throw new Error(`HTTP Error: ${response.status}`);
        }

        const html = await response.text();

        const $ = cheerio.load(html);

        const companies = [];

        // Hämtar ENDAST företagsnamn från h2.MuiTypography-h2
        $("h2.MuiTypography-h2").each((index, element) => {
            const companyName = $(element)
                .text()
                .trim();

            if (companyName) {
                companies.push(companyName);
            }
        });


        // Skapa CSV
        let csv = "Företagsnamn\n";

        companies.forEach(name => {
            csv += `"${name.replace(/"/g, '""')}"\n`;
        });


        fs.writeFileSync(
            "rubriker.csv",
            csv,
            "utf8"
        );


        console.log(
            `Klart! ${companies.length} företagsnamn sparades i rubriker.csv`
        );


    } catch (error) {
        console.error("Fel:", error.message);
    }
}


scrapeCompanies();
